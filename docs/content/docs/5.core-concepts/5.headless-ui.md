# Headless & Custom UI

Notiq decouples its Lexical engine, CRDT collaboration, AI streaming/thinking/research pipeline, and block commands from its default React/Tailwind components. You can run every layer headlessly and supply your own design system, Radix/Ark/shadcn primitives, or plain HTML.

---

## 1. Full End-to-End Headless Example

Below is a complete React editor that uses **zero** default Notiq menus, AI bars, or callout/poll UI—while keeping full keyboard shortcuts, `/` filtering, drag-and-drop block handles, `Cmd+K` AI streaming, and inline ghost-text completions:

```tsx
"use client";

import React from "react";
import {
  StarterKit,
  BlocksKit,
  AICopilot,
} from "@collabchron/notiq";
import {
  useNotiq,
  NotiqContent,
  BubbleMenu,
  FloatingMenu,
  type HintRendererProps,
  type PollRendererProps,
} from "@collabchron/notiq/react";

function CustomHint({ type, emoji }: HintRendererProps) {
  return (
    <aside
      data-hint-variant={type}
      style={{
        borderLeft: "4px solid currentColor",
        padding: "12px 16px",
        borderRadius: 8,
        margin: "12px 0",
      }}
    >
      <strong>{emoji ?? "💡"} {type.toUpperCase()}</strong>
    </aside>
  );
}

function CustomPoll({ question, options }: PollRendererProps) {
  return (
    <div style={{ border: "1px solid #333", padding: 12, borderRadius: 8 }}>
      <p style={{ fontWeight: 600 }}>{question}</p>
      <ul>
        {options.map((opt) => (
          <li key={opt.uid}>
            {opt.text} ({opt.votes.length} votes)
          </li>
        ))}
      </ul>
    </div>
  );
}

export function HeadlessEditor() {
  const editor = useNotiq({
    extensions: [
      StarterKit,
      BlocksKit.configure({
        headless: true,
        slashMenu: {
          render: ({
            options,
            selectedIndex,
            queryString,
            selectOptionAndCleanUp,
            setHighlightedIndex,
          }) => (
            <div role="listbox" aria-label="Commands">
              <div>Filter: {queryString ?? "all"}</div>
              {options.map((option, index) => (
                <button
                  key={option.key}
                  type="button"
                  data-selected={index === selectedIndex}
                  onMouseEnter={() => setHighlightedIndex(index)}
                  onClick={() => selectOptionAndCleanUp(option)}
                >
                  {option.title}
                </button>
              ))}
            </div>
          ),
        },
        handles: {
          render: ({
            menuRef,
            isMenuOpen,
            setIsMenuOpen,
            insertBelow,
            turnInto,
            duplicate,
            remove,
            askAI,
          }) => (
            <div ref={menuRef} className="draggable-block-menu">
              <button type="button" onClick={insertBelow}>+</button>
              <button type="button" onClick={() => setIsMenuOpen((v) => !v)}>
                ⋮⋮
              </button>
              {isMenuOpen && (
                <div>
                  <button type="button" onClick={askAI}>Ask AI</button>
                  <button type="button" onClick={() => turnInto("h2")}>H2</button>
                  <button type="button" onClick={duplicate}>Duplicate</button>
                  <button type="button" onClick={remove}>Delete</button>
                </div>
              )}
            </div>
          ),
          renderTargetLine: (targetLineRef) => (
            <div
              ref={targetLineRef}
              style={{ height: 2, background: "#6366f1", position: "absolute" }}
            />
          ),
        },
        renderers: {
          hint: CustomHint,
          poll: CustomPoll,
        },
      }),
      AICopilot.configure({
        headless: true,
        apiEndpoint: "/api/ai",
        render: ({ suggestion, accept, dismiss }) =>
          suggestion ? (
            <span data-ghost-text>
              {suggestion}
              <button type="button" onClick={accept}>Tab</button>
              <button type="button" onClick={dismiss}>Esc</button>
            </span>
          ) : null,
        commandBar: {
          headless: true,
          render: ({
            isOpen,
            prompt,
            setPrompt,
            submit,
            status,
            thinking,
            sources,
            accept,
            discard,
            close,
          }) =>
            isOpen ? (
              <div role="dialog" aria-label="AI Assistant">
                <input
                  value={prompt}
                  onChange={(e) => setPrompt(e.target.value)}
                  placeholder="Ask AI..."
                />
                <button type="button" onClick={() => submit()}>
                  Generate ({status})
                </button>
                {thinking && <pre>{thinking}</pre>}
                {sources.length > 0 && (
                  <ul>
                    {sources.map((s) => (
                      <li key={s.url}>
                        <a href={s.url}>{s.title}</a>
                      </li>
                    ))}
                  </ul>
                )}
                {status === "review" && (
                  <>
                    <button type="button" onClick={accept}>Accept</button>
                    <button type="button" onClick={discard}>Discard</button>
                  </>
                )}
                <button type="button" onClick={close}>Close</button>
              </div>
            ) : null,
        },
      }),
    ],
  });

  return (
    <div className="my-editor-shell">
      <BubbleMenu editor={editor}>
        <button type="button" onClick={() => editor?.chain().focus().toggleBold().run()}>
          Bold
        </button>
        <button type="button" onClick={() => editor?.chain().focus().toggleItalic().run()}>
          Italic
        </button>
      </BubbleMenu>

      <FloatingMenu editor={editor}>
        <button type="button" onClick={() => editor?.chain().focus().setHeading({ level: 1 }).run()}>
          H1
        </button>
      </FloatingMenu>

      <NotiqContent
        editor={editor}
        className="my-editable"
        placeholder="Type '/' for commands or Cmd+K for AI..."
      />
    </div>
  );
}
```

## 2. External State Sync (`onStateChange`)

If your UI lives outside the editor React tree (for example in a sidebar, status bar, or Zustand/Redux store), pass `headless: true` with `onStateChange` instead of `render`:

- **`SlashMenu.configure({ headless: true, onStateChange })`**: receives `{ isOpen, options, selectedIndex, queryString, selectOptionAndCleanUp, setHighlightedIndex }`.
- **`BlockHandle.configure({ headless: true, onStateChange })`**: receives `{ menuRef, targetLineRef, isMenuOpen, setIsMenuOpen, insertBelow, turnInto, splitColumns, duplicate, remove, askAI }`.
- **`AICommandBar.configure({ headless: true, onStateChange })`**: receives `{ isOpen, prompt, status, thinking, sources, submit, accept, discard, close }`.
- **`AICopilot.configure({ headless: true, onStateChange })`**: receives `{ enabled, suggestion, isLoading, toggle, accept, dismiss }`.

## 3. Custom Node Renderers

Rich decorator nodes (`hint`, `excalidraw`, `poll`, `image`, `inlineImage`) can be overridden in three ways:

1. **Extension-level**: `BlocksKit.configure({ renderers: { hint: MyHint } })`
2. **Content-level**: `<NotiqContent editor={editor} renderers={{ poll: MyPoll }} />`
3. **Provider-level**: `<NotiqNodeRendererProvider renderers={{ image: MyImage }}>...</NotiqNodeRendererProvider>`

Any node type without a custom override automatically falls back to Notiq's built-in component.
